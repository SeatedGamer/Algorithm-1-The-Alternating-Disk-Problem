ALGORITHM make_alternating_disks(n)
  // Input: positive integer n
  // Output: list of 2n disks alternating dark, light, starting with dark

  disks ← list of size 2n
  for i ← 0 to 2n − 1 do
      if i is even then disks[i] ← DARK
      else disks[i] ← LIGHT
  return disks


ALGORITHM sort_disks(disks)
  // Input: list of 2n alternating disks
  // Output: disks rearranged so all dark disks are on the left and all
  //         light disks are on the right; returns the number of passes

  size ← length of disks
  swaps ← 0
  passes ← 0
  swapped ← true

  while swapped do
      swapped ← false

      // Left-to-right pass
      for i ← 0 to size − 2 do
          if disks[i] = LIGHT and disks[i + 1] = DARK then
              swap disks[i] and disks[i + 1]
              swaps ← swaps + 1
              swapped ← true
      passes ← passes + 1
      print "Pass", passes, "(L->R):", disks

      // Right-to-left pass
      for i ← size − 2 down to 0 do
          if disks[i] = LIGHT and disks[i + 1] = DARK then
              swap disks[i] and disks[i + 1]
              swaps ← swaps + 1
              swapped ← true
      passes ← passes + 1
      print "Pass", passes, "(R->L):", disks

  return passes
